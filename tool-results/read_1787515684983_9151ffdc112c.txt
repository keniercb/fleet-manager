    77→    77→    public ReporteMovimientoMensualResponse reporteMovimientoMensual(Long vehiculoId, Integer mes, Integer anio) {
    78→    78→        if (mes < 1 || mes > 12) {
    79→    79→            throw new BusinessException("El mes debe estar entre 1 y 12");
    80→    80→        }
    81→    81→
    82→    82→        Vehiculo vehiculo = vehiculoRepository.findById(vehiculoId)
    83→    83→                .orElseThrow(() -> new ResourceNotFoundException("Vehiculo", "id", vehiculoId));
    84→    84→
    85→    85→        YearMonth yearMonth = YearMonth.of(anio, mes);
    86→    86→        LocalDate firstDay = yearMonth.atDay(1);
    87→    87→        LocalDate lastDay = yearMonth.atEndOfMonth();
    88→    88→        int daysInMonth = yearMonth.lengthOfMonth();
    89→    89→
    90→    90→        // Obtener todos los recorridos del vehiculo hasta el ultimo dia del mes
    91→    91→        List<Recorrido> recorridos = repository.findByVehiculoIdAndFechaLessThanEqualOrderByFechaAsc(vehiculoId, lastDay);
    92→    92→
    93→    93→        // Map de recorridos del mes: dia -> recorrido
    94→    94→        Map<Integer, Recorrido> recorridosDelMes = recorridos.stream()
    95→    95→                .filter(r -> !r.getFecha().isBefore(firstDay) && !r.getFecha().isAfter(lastDay))
    96→    96→                .collect(Collectors.toMap(r -> r.getFecha().getDayOfMonth(), r -> r));
    97→    97→
    98→    98→        // Establecer estado inicial a partir de recorridos anteriores al mes
    99→    99→        BigDecimal runningCombustible = CERO;
   100→   100→        BigInteger runningOdometro = BigInteger.valueOf(0L);
   101→   101→        boolean estadoInicialCalculado = false;
   102→   102→
   103→   103→        for (Recorrido r : recorridos) {
   104→   104→            if (r.getFecha().isBefore(firstDay)) {
   105→   105→                runningOdometro = r.getOdometroInicial().add(BigInteger.valueOf(r.getKilometros()));
   106→   106→                runningCombustible = r.getCombustibleInicial()
   107→   107→                        .subtract(r.getConsumo() != null ? r.getConsumo() : CERO)
   108→   108→                        .add(r.getLitrosAbastecidos() != null ? r.getLitrosAbastecidos() : CERO);
   109→   109→                estadoInicialCalculado = true;
   110→   110→            }
   111→   111→        }
   112→   112→
   113→   113→        // Construir lecturas diarias
   114→   114→        List<LecturaDiariaResponse> lecturas = new ArrayList<>();
   115→   115→        BigDecimal totalConsumido = CERO;
   116→   116→        BigDecimal totalAbastecido = CERO;
   117→   117→        int totalKilometros = 0;
   118→   118→        BigDecimal primerCombustibleEnDeposito = null;
   119→   119→
   120→   120→        for (int dia = 1; dia <= daysInMonth; dia++) {
   121→   121→            Recorrido recorrido = recorridosDelMes.get(dia);
   122→   122→
   123→   123→            BigInteger odometro;
   124→   124→            BigDecimal combustibleEnDeposito;
   125→   125→            BigDecimal combustibleConsumido;
   126→   126→            BigDecimal combustibleAbastecido;
   127→   127→
   128→   128→            if (recorrido != null) {
   129→   129→                odometro = recorrido.getOdometroInicial();
   130→   130→                combustibleEnDeposito = recorrido.getCombustibleInicial();
   131→   131→                combustibleConsumido = recorrido.getConsumo() != null ? recorrido.getConsumo() : CERO;
   132→   132→                combustibleAbastecido = recorrido.getLitrosAbastecidos() != null ? recorrido.getLitrosAbastecidos() : CERO;
   133→   133→
   134→   134→                totalConsumido = totalConsumido.add(combustibleConsumido);
   135→   135→                totalAbastecido = totalAbastecido.add(combustibleAbastecido);
   136→   136→                totalKilometros += recorrido.getKilometros();
   137→   137→
   138→   138→                // Actualizar estado corriendo
   139→   139→                runningOdometro = odometro.add(BigInteger.valueOf(recorrido.getKilometros()));
   140→   140→                runningCombustible = combustibleEnDeposito.subtract(combustibleConsumido).add(combustibleAbastecido);
   141→   141→            } else {
   142→   142→                odometro = runningOdometro;
   143→   143→                combustibleEnDeposito = runningCombustible;
   144→   144→                combustibleConsumido = CERO;
   145→   145→                combustibleAbastecido = CERO;
   146→   146→            }
   147→   147→
   148→   148→            if (primerCombustibleEnDeposito == null) {
   149→   149→                primerCombustibleEnDeposito = combustibleEnDeposito;
   150→   150→            }
   151→   151→
   152→   152→            BigDecimal saldoCombustible = combustibleEnDeposito.add(combustibleAbastecido).subtract(combustibleConsumido);
   153→   153→
   154→   154→            int kilometrosRecorridos = recorrido != null ? recorrido.getKilometros() : 0;
   155→   155→
   156→   156→            lecturas.add(LecturaDiariaResponse.builder()
   157→   157→                    .dia(dia)
   158→   158→                    .odometro(odometro)
   159→   159→                    .combustibleEnDeposito(combustibleEnDeposito)
   160→   160→                    .combustibleConsumido(combustibleConsumido)
   161→   161→                    .combustibleAbastecido(combustibleAbastecido)
   162→   162→                    .saldoCombustible(saldoCombustible)
   163→   163→                    .kilometrosRecorridos(kilometrosRecorridos)
   164→   164→                    .build());
   165→   165→        }
   166→   166→
   167→   167→        // Analisis de consumo
   168→   168→        BigDecimal existenciaFinal = primerCombustibleEnDeposito != null
   169→   169→                ? primerCombustibleEnDeposito.subtract(totalConsumido).add(totalAbastecido)
   170→   170→                : CERO;
   171→   171→        BigDecimal consumidoSegunNorma = vehiculo.getIndiceConsumo()
   172→   172→                .multiply(BigDecimal.valueOf(totalKilometros))
   173→   173→                .divide(CIEN, 2, RoundingMode.HALF_UP);
   174→   174→
   175→   175→        AnalisisConsumoResponse analisis = AnalisisConsumoResponse.builder()
   176→   176→                .combustibleInicial(primerCombustibleEnDeposito != null ? primerCombustibleEnDeposito : CERO)
   177→   177→                .combustibleRecibido(totalAbastecido)
   178→   178→                .combustibleConsumido(totalConsumido)
   179→   179→                .existenciaFinal(existenciaFinal)
   180→   180→                .kilometrosRecorridos(totalKilometros)
   181→   181→                .consumidoSegunNorma(consumidoSegunNorma)
   182→   182→                .build();
   183→   183→
   184→   184→        // Datos del vehiculo
   185→   185→        ChoferResponse choferResp = vehiculo.getChofer() != null ? toChoferResumido(vehiculo.getChofer()) : null;
   186→   186→        VehiculoReporteData vehiculoData = VehiculoReporteData.builder()
   187→   187→                .marca(vehiculo.getMarca().getNombre())
   188→   188→                .numeroMotor(vehiculo.getNumeroMotor())
   189→   189→                .tipoCombustible(vehiculo.getTipoCombustible().getDenominacion())
   190→   190→                .normaConsumo(vehiculo.getIndiceConsumo())

   192→   192→                .chofer(choferResp)
   193→   193→                .build();
   194→   194→
   195→   195→        return ReporteMovimientoMensualResponse.builder()
   196→   196→                .vehiculo(vehiculoData)
   197→   197→                .lecturas(lecturas)
   198→   198→                .analisis(analisis)
   199→   199→                .build();
   200→   200→    }

   202→   202→    @Override
   203→   203→    @Transactional
   204→   204→    public RecorridoResponse create(RecorridoRequest request) {
   205→   205→        Vehiculo vehiculo = vehiculoRepository.findById(request.getVehiculoId())
   206→   206→                .orElseThrow(() -> new ResourceNotFoundException("Vehiculo", "id", request.getVehiculoId()));
   207→   207→
   208→   208→        // Validar unicidad: solo un recorrido por vehiculo por fecha
   209→   209→        if (repository.existsByVehiculoIdAndFecha(request.getVehiculoId(), request.getFecha())) {
   210→   210→            throw new BusinessException("Ya existe un recorrido para el vehiculo con id "
   211→   211→                    + request.getVehiculoId() + " en la fecha " + request.getFecha());
   212→   212→        }
   213→   213→
   214→   214→        // Validar que no exista un recorrido con fecha posterior para el mismo vehiculo
   215→   215→        if (repository.existsByVehiculoIdAndFechaAfter(request.getVehiculoId(), request.getFecha())) {
   216→   216→            throw new BusinessException("No se puede insertar el recorrido en la fecha "
   217→   217→                    + request.getFecha() + " porque ya existe un recorrido con fecha posterior para el vehiculo con id "
   218→   218→                    + request.getVehiculoId());
   219→   219→        }
   220→   220→
   221→   221→        // Calcular consumo: (indiceConsumo * kilometros) / 100, redondeado a 2 decimales
   222→   222→        BigDecimal consumo = vehiculo.getIndiceConsumo()
   223→   223→                .multiply(BigDecimal.valueOf(request.getKilometros()))
   224→   224→                .divide(CIEN, 2, RoundingMode.HALF_UP);
   225→   225→
   226→   226→        // Redondear litros abastecidos a 2 decimales
   227→   227→        BigDecimal litrosAbastecidos = request.getLitrosAbastecidos() != null
   228→   228→                ? request.getLitrosAbastecidos().setScale(2, RoundingMode.HALF_UP)
   229→   229→                : CERO;
   230→   230→
   231→   231→        // Validar que el vehiculo tenga suficiente combustible (sumando litros abastecidos)
   232→   232→        BigDecimal combustibleRestante = vehiculo.getCombustible().subtract(consumo).add(litrosAbastecidos);
   233→   233→        if (combustibleRestante.compareTo(BigDecimal.ZERO) < 0) {
   234→   234→            throw new BusinessException("El recorrido no puede ser insertado porque se consume mas combustible ("
   235→   235→                    + consumo + ") que el disponible en el vehiculo (" + vehiculo.getCombustible() + ")");
   236→   236→        }
   237→   237→
   238→   238→        // Resolver chofer: si no se envia, usar el chofer asignado al vehiculo
   239→   239→        Chofer chofer = resolverChofer(request.getChoferId(), vehiculo);
   240→   240→
   241→   241→        // Validar tarjeta de combustible y descontar importe
   242→   242→        TarjetaCombustible tarjetaCombustible = null;
   243→   243→        if (request.getTarjetaCombustibleId() != null) {
   244→   244→            if (request.getImporteAbastecido() == null || request.getImporteAbastecido() <= 0) {
   245→   245→                throw new BusinessException("Si se envia tarjeta de combustible, el importe abastecido es obligatorio y debe ser mayor a cero");
   246→   246→            }
   247→   247→            tarjetaCombustible = tarjetaCombustibleRepository.findById(request.getTarjetaCombustibleId())
   248→   248→                    .orElseThrow(() -> new ResourceNotFoundException("TarjetaCombustible", "id", request.getTarjetaCombustibleId()));
   249→   249→            if (tarjetaCombustible.getSaldo() <= request.getImporteAbastecido()) {
   250→   250→                throw new BusinessException("El saldo de la tarjeta no puede quedar en cero o negativo tras el descuento");
   251→   251→            }
   252→   252→            tarjetaCombustible.setSaldo(tarjetaCombustible.getSaldo() - request.getImporteAbastecido());
   253→   253→            tarjetaCombustibleRepository.save(tarjetaCombustible);
   254→   254→        } else if (request.getImporteAbastecido() != null) {
   255→   255→            throw new BusinessException("Si se envia importe abastecido, debe enviarse la tarjeta de combustible");
   256→   256→        }
   257→   257→
   258→   258→        // OdometroInicial es el odometro actual del vehiculo antes de sumar kilometros
   259→   259→        BigInteger odometroInicial = vehiculo.getOdometro();
   260→   260→
   261→   261→        // CombustibleInicial es el combustible actual del vehiculo antes del recorrido
   262→   262→        BigDecimal combustibleInicial = vehiculo.getCombustible();
   263→   263→
   264→   264→        Recorrido entity = Recorrido.builder()
   265→   265→                .vehiculo(vehiculo)
   266→   266→                .chofer(chofer)
   267→   267→                .fecha(request.getFecha())
   268→   268→                .kilometros(request.getKilometros())
   269→   269→                .odometroInicial(odometroInicial)
   270→   270→                .combustibleInicial(combustibleInicial)
   271→   271→                .consumo(consumo)
   272→   272→                .litrosAbastecidos(litrosAbastecidos)
   273→   273→                .numeroChip(request.getNumeroChip())
   274→   274→                .lugarAbastecimiento(request.getLugarAbastecimiento())
   275→   275→                .tarjetaCombustible(tarjetaCombustible)
   276→   276→                .importeAbastecido(request.getImporteAbastecido())
   277→   277→                .activo(true)
   278→   278→                .build();
   279→   279→
   280→   280→        Recorrido saved = repository.save(entity);
   281→   281→
   282→   282→        // Sumar kilometros al odometro del vehiculo
   283→   283→        vehiculo.setOdometro(vehiculo.getOdometro().add(BigInteger.valueOf(request.getKilometros())));
   284→   284→        // Actualizar combustible del vehiculo
   285→   285→        vehiculo.setCombustible(combustibleRestante);
   286→   286→        vehiculoRepository.save(vehiculo);
   287→   287→
   288→   288→        return toResponse(saved);
   289→   289→    }

   291→   291→    @Override
   292→   292→    @Transactional
   293→   293→    public RecorridoResponse update(Long id, RecorridoRequest request) {
   294→   294→        Recorrido entity = repository.findById(id)
   295→   295→                .orElseThrow(() -> new ResourceNotFoundException("Recorrido", "id", id));
   296→   296→
   297→   297→        // No se permite cambiar el vehiculo
   298→   298→        if (!entity.getVehiculo().getId().equals(request.getVehiculoId())) {
   299→   299→            throw new BusinessException("No se permite cambiar el vehiculo del recorrido");
   300→   300→        }
   301→   301→
   302→   302→        // No se permite cambiar la fecha
   303→   303→        if (!entity.getFecha().equals(request.getFecha())) {
   304→   304→            throw new BusinessException("No se permite cambiar la fecha del recorrido");
   305→   305→        }
   306→   306→
   307→   307→        Vehiculo vehiculo = entity.getVehiculo();
   308→   308→
   309→   309→        // No se permite modificar si existe un recorrido con fecha posterior
   310→   310→        if (repository.existsByVehiculoIdAndFechaAfter(vehiculo.getId(), entity.getFecha())) {
   311→   311→            throw new BusinessException("No se puede modificar el recorrido porque existe un recorrido con fecha posterior para el vehiculo con id "
   312→   312→                    + vehiculo.getId());
   313→   313→        }
   314→   314→
   315→   315→        // Calcular el nuevo consumo, redondeado a 2 decimales
   316→   316→        BigDecimal nuevoConsumo = vehiculo.getIndiceConsumo()
   317→   317→                .multiply(BigDecimal.valueOf(request.getKilometros()))
   318→   318→                .divide(CIEN, 2, RoundingMode.HALF_UP);
   319→   319→
   320→   320→        // Verificar disponibilidad de combustible restaurando el consumo antiguo primero
   321→   321→        BigDecimal consumoAntiguo = entity.getConsumo() != null ? entity.getConsumo() : CERO;
   322→   322→        BigDecimal combustibleDisponible = vehiculo.getCombustible().add(consumoAntiguo).subtract(nuevoConsumo);
   323→   323→        if (combustibleDisponible.compareTo(BigDecimal.ZERO) < 0) {
   324→   324→            throw new BusinessException("El recorrido no puede ser actualizado porque se consume mas combustible ("
   325→   325→                    + nuevoConsumo + ") que el disponible en el vehiculo ("
   326→   326→                    + vehiculo.getCombustible().add(consumoAntiguo) + ")");
   327→   327→        }
   328→   328→
   329→   329→        // Restaurar odometro y combustible del vehiculo
   330→   330→        BigInteger kilometrosAnteriores = BigInteger.valueOf(entity.getKilometros());
   331→   331→        vehiculo.setOdometro(vehiculo.getOdometro().subtract(kilometrosAnteriores));
   332→   332→        vehiculo.setCombustible(vehiculo.getCombustible().add(consumoAntiguo));
   333→   333→
   334→   334→        // Resolver chofer: si no se envia, usar el chofer asignado al vehiculo
   335→   335→        Chofer chofer = resolverChofer(request.getChoferId(), vehiculo);
   336→   336→
   337→   337→        // Manejar tarjeta de combustible: restablecer saldo anterior y descontar nuevo importe
   338→   338→        TarjetaCombustible tarjetaCombustible = null;
   339→   339→        TarjetaCombustible tarjetaAnterior = entity.getTarjetaCombustible();
   340→   340→        Double importeAnterior = entity.getImporteAbastecido() != null ? entity.getImporteAbastecido() : 0.0;
   341→   341→
   342→   342→        if (request.getTarjetaCombustibleId() != null) {
   343→   343→            if (request.getImporteAbastecido() == null || request.getImporteAbastecido() <= 0) {
   344→   344→                throw new BusinessException("Si se envia tarjeta de combustible, el importe abastecido es obligatorio y debe ser mayor a cero");
   345→   345→            }
   346→   346→            tarjetaCombustible = tarjetaCombustibleRepository.findById(request.getTarjetaCombustibleId())
   347→   347→                    .orElseThrow(() -> new ResourceNotFoundException("TarjetaCombustible", "id", request.getTarjetaCombustibleId()));
   348→   348→
   349→   349→            Double nuevoImporte = request.getImporteAbastecido();
   350→   350→
   351→   351→            // Restablecer saldo de tarjeta anterior si existe
   352→   352→            if (tarjetaAnterior != null) {
   353→   353→                if (tarjetaAnterior.getId().equals(tarjetaCombustible.getId())) {
   354→   354→                    // Misma tarjeta: restaurar importe anterior al saldo actual
   355→   355→                    tarjetaCombustible.setSaldo(tarjetaCombustible.getSaldo() + importeAnterior);
   356→   356→                } else {
   357→   357→                    // Tarjeta diferente: restaurar en la anterior
   358→   358→                    tarjetaAnterior.setSaldo(tarjetaAnterior.getSaldo() + importeAnterior);
   359→   359→                    tarjetaCombustibleRepository.save(tarjetaAnterior);
   360→   360→                }
   361→   361→            }
   362→   362→
   363→   363→            // Validar y restar nuevo importe
   364→   364→            if (tarjetaCombustible.getSaldo() <= nuevoImporte) {
   365→   365→                throw new BusinessException("El saldo de la tarjeta no puede quedar en cero o negativo tras el descuento");
   366→   366→            }
   367→   367→            tarjetaCombustible.setSaldo(tarjetaCombustible.getSaldo() - nuevoImporte);
   368→   368→            tarjetaCombustibleRepository.save(tarjetaCombustible);
   369→   369→        } else if (request.getImporteAbastecido() != null) {
   370→   370→            throw new BusinessException("Si se envia importe abastecido, debe enviarse la tarjeta de combustible");
   371→   371→        } else if (tarjetaAnterior != null) {
   372→   372→            // Se elimino la tarjeta: restablecer saldo de la tarjeta anterior
   373→   373→            tarjetaAnterior.setSaldo(tarjetaAnterior.getSaldo() + importeAnterior);
   374→   374→            tarjetaCombustibleRepository.save(tarjetaAnterior);
   375→   375→        }
   376→   376→
   377→   377→        // Actualizar la entidad
   378→   378→        entity.setChofer(chofer);
   379→   379→        entity.setKilometros(request.getKilometros());
   380→   380→        entity.setOdometroInicial(vehiculo.getOdometro());
   381→   381→        entity.setConsumo(nuevoConsumo);
   382→   382→        entity.setTarjetaCombustible(tarjetaCombustible);
   383→   383→        entity.setImporteAbastecido(request.getImporteAbastecido());
   384→   384→
   385→   385→        Recorrido saved = repository.save(entity);
   386→   386→
   387→   387→        // Sumar kilometros al odometro y restar consumo al combustible del vehiculo
   388→   388→        vehiculo.setOdometro(vehiculo.getOdometro().add(BigInteger.valueOf(request.getKilometros())));
   389→   389→        vehiculo.setCombustible(vehiculo.getCombustible().subtract(nuevoConsumo));
   390→   390→        vehiculoRepository.save(vehiculo);
   391→   391→
   392→   392→        return toResponse(saved);
   393→   393→    }

   395→   395→    @Override
   396→   396→    @Transactional
   397→   397→    public void delete(Long id) {
   398→   398→        Recorrido entity = repository.findById(id)
   399→   399→                .orElseThrow(() -> new ResourceNotFoundException("Recorrido", "id", id));
   400→   400→
   401→   401→        Vehiculo vehiculo = entity.getVehiculo();
   402→   402→
   403→   403→        // No se permite eliminar si existe un recorrido con fecha posterior
   404→   404→        if (repository.existsByVehiculoIdAndFechaAfter(vehiculo.getId(), entity.getFecha())) {
   405→   405→            throw new BusinessException("No se puede eliminar el recorrido porque existe un recorrido con fecha posterior para el vehiculo con id "
   406→   406→                    + vehiculo.getId());
   407→   407→        }
   408→   408→
   409→   409→        // Restar los kilometros al odometro del vehiculo
   410→   410→        BigInteger kilometros = BigInteger.valueOf(entity.getKilometros());
   411→   411→        vehiculo.setOdometro(vehiculo.getOdometro().subtract(kilometros));
   412→   412→        // Restaurar el combustible consumido y los litros abastecidos
   413→   413→        BigDecimal consumo = entity.getConsumo() != null ? entity.getConsumo() : CERO;
   414→   414→        BigDecimal litrosAbastecidos = entity.getLitrosAbastecidos() != null ? entity.getLitrosAbastecidos() : CERO;
   415→   415→        vehiculo.setCombustible(vehiculo.getCombustible().add(consumo).subtract(litrosAbastecidos));
   416→   416→        vehiculoRepository.save(vehiculo);
   417→   417→
   418→   418→        // Restablecer saldo de tarjeta de combustible si existe
   419→   419→        if (entity.getTarjetaCombustible() != null && entity.getImporteAbastecido() != null) {
   420→   420→            TarjetaCombustible tarjeta = entity.getTarjetaCombustible();
   421→   421→            tarjeta.setSaldo(tarjeta.getSaldo() + entity.getImporteAbastecido());
   422→   422→            tarjetaCombustibleRepository.save(tarjeta);
   423→   423→        }
   424→   424→
   425→   425→        // Eliminacion fisica
   426→   426→        repository.delete(entity);
   427→   427→    }
